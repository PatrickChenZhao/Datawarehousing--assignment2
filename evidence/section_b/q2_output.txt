SQL> WITH rollup_sales AS (
  2      SELECT CASE
  3                 WHEN GROUPING(c.country_region) = 1 THEN 'ALL REGIONS'
  4                 ELSE c.country_region
  5             END AS country_region,
  6             CASE
  7                 WHEN GROUPING(c.country_region) = 1
  8                  AND GROUPING(c.country_name) = 1 THEN 'GRAND TOTAL'
  9                 WHEN GROUPING(c.country_name) = 1 THEN 'REGION TOTAL'
 10                 ELSE c.country_name
 11             END AS country_name,
 12             ROUND(SUM(s.amount_sold)) AS rounded_sales,
 13             GROUPING_ID(c.country_region, c.country_name) AS group_level
 14      FROM sh.sales s
 15      JOIN sh.products p ON p.prod_id = s.prod_id
 16      JOIN sh.customers cu ON cu.cust_id = s.cust_id
 17      JOIN sh.countries c ON c.country_id = cu.country_id
 18      WHERE c.country_region = 'Europe'
 19        AND p.prod_subcategory = 'Recordable CDs'
 20      GROUP BY ROLLUP(c.country_region, c.country_name)
 21  )
 22  SELECT country_region AS COUNTRY_REGION,
 23         country_name AS COUNTRY_NAME,
 24         TO_CHAR(
 25             rounded_sales,
 26             'FM$999,999,999,999,990',
 27             'NLS_NUMERIC_CHARACTERS=''.,'''
 28         ) || '.xx' AS TOTAL_SALES
 29  FROM rollup_sales
